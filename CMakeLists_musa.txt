cmake_minimum_required(VERSION 3.18)

project(
    tiny-cuda-nn
    VERSION 2.0
    DESCRIPTION "tiny-cuda-nn MUSA port (encodings-only)"
    LANGUAGES CXX
)

# ── MUSA toolchain ─────────────────────────────────────────────────────────────
list(APPEND CMAKE_MODULE_PATH /usr/local/musa/cmake)
find_package(MUSA REQUIRED)

set(MUSA_MCC_FLAGS
    --offload-arch=mp_31       # S5000 (MT-3000 series, major=3 minor=1)
    -Wno-unused-function
    -Wno-unused-variable
)

# ── Options ────────────────────────────────────────────────────────────────────
option(TCNN_BUILD_BENCHMARK "Build benchmarks?" OFF)
option(TCNN_BUILD_EXAMPLES  "Build examples?"   OFF)
option(TCNN_BUILD_TESTS     "Build tests?"      OFF)

# ── C++ standard ──────────────────────────────────────────────────────────────
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# ── Sources ──────────────────────────────────────────────────────────────────
# Round 2: cutlass_mlp.mu + rtc_kernel.mu enabled via MUSA-native backings
#   cutlass_mlp.mu        — muBLAS shim (include_musa/tiny-cuda-nn/cutlass_matmul.h)
#   rtc_kernel.mu         — mtrtc + muModuleLoadFatBinary
#   fully_fused_mlp.mu    — still excluded; needs nvcuda::wmma → mma equivalent
#                           (scheduled as a narrower follow-up; blocked on
#                            mcc WMMA overload availability verification)
set(TCNN_SOURCES
    src_musa/common_host.mu
    src_musa/cpp_api.mu
    src_musa/cutlass_mlp.mu       # Round 2: real port via muBLAS shim
    src_musa/encoding.mu
    src_musa/loss.mu
    src_musa/network.mu
    src_musa/object.mu
    src_musa/optimizer.mu
    src_musa/reduce_sum.mu
    src_musa/rtc_kernel.mu        # Round 2: nvrtc → mtrtc + fatbin module-load
    # fully_fused_mlp.mu excluded: depends on nvcuda::wmma PTX
)

# ── Definitions ───────────────────────────────────────────────────────────────
set(TCNN_DEFINITIONS
    -DTCNN_MIN_GPU_ARCH=21        # mp_21 = S5000
    -DTCNN_HALF_PRECISION=1
    -DTCNN_NO_FULLY_FUSED_MLP     # disable wmma code paths
    -DFMT_USE_FLOAT128=0          # SDK 5.1.0 (Round 1): mcc mp_31 rejects __float128
    -DTCNN_RTC                    # Round 2: enable runtime compilation via mtrtc
    # TCNN_SHAMPOO not defined      # disable cuBLAS/cuSolver Shampoo optimizer
)

# ── Includes ──────────────────────────────────────────────────────────────────
set(TCNN_INCLUDES
    include_musa                  # ported headers
    dependencies                  # fmt, json, pcg32, cmrc (CUDA-free)
    # dependencies/cutlass/include NOT included (NVIDIA-specific)
)

# ── fmt dependency (CUDA-free) ────────────────────────────────────────────────
set(BUILD_SHARED_LIBS OFF)
set(FMT_UNICODE OFF CACHE BOOL "" FORCE)
add_subdirectory("dependencies/fmt")

# ── cmrc resource library ─────────────────────────────────────────────────────
include("${CMAKE_CURRENT_SOURCE_DIR}/dependencies/cmrc/CMakeRC.cmake")
cmrc_add_resource_library(tiny-cuda-nn-resources NAMESPACE tcnn)
set(TCNN_DEFINITIONS ${TCNN_DEFINITIONS} -DTCNN_CMRC)

# ── Build the library ──────────────────────────────────────────────────────────
musa_add_library(tiny-cuda-nn STATIC ${TCNN_SOURCES})
target_compile_definitions(tiny-cuda-nn PUBLIC ${TCNN_DEFINITIONS})
target_include_directories(tiny-cuda-nn PUBLIC ${TCNN_INCLUDES})
# Round 2: link mublas (CUTLASS shim backend) + mtrtc (JIT) + musart/musa driver.
# FindMUSA does not auto-propagate `-L/usr/local/musa/lib` to downstream targets;
# an explicit target_link_directories is required or ld rejects the bare library
# names ("cannot find -lmublas").
target_link_directories(tiny-cuda-nn PUBLIC /usr/local/musa/lib)
target_link_libraries(tiny-cuda-nn
    fmt
    tiny-cuda-nn-resources
    mublas
    mtrtc
    musart
    musa
)

# ── Tests ─────────────────────────────────────────────────────────────────────
if (TCNN_BUILD_TESTS)
    enable_testing()
    add_subdirectory(tests_musa)
endif()
